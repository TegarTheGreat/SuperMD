<script>
(function () {
  "use strict";

  var STATUSES = ["paid", "pending", "shipped", "refunded", "cancelled"];
  var CUSTOMERS = [
    ["Ada Lovelace", "ada@example.com"],
    ["Grace Hopper", "grace@example.com"],
    ["Alan Turing", "alan@example.com"],
    ["Katherine Johnson", "katherine@example.com"],
    ["Linus Torvalds", "linus@example.com"],
    ["Margaret Hamilton", "margaret@example.com"],
    ["Barbara Liskov", "barbara@example.com"],
    ["Ken Thompson", "ken@example.com"],
    ["Radia Perlman", "radia@example.com"],
    ["Donald Knuth", "donald@example.com"],
    ["Anita Borg", "anita@example.com"],
    ["Tim Berners-Lee", "tim@example.com"]
  ];
  var PRODUCTS = [
    ["Mechanical Keyboard", 129.0],
    ["USB-C Hub", 49.5],
    ["27\" Monitor", 319.0],
    ["Noise-Cancelling Headphones", 249.0],
    ["Ergonomic Mouse", 79.0],
    ["Laptop Stand", 39.0],
    ["Webcam 1080p", 89.0],
    ["Desk Lamp", 45.0],
    ["External SSD 1TB", 139.0],
    ["Cable Organizer", 15.0]
  ];

  // Deterministic pseudo-random so the sample data is stable across reloads.
  var seed = 20240517;
  function rand() {
    seed = (seed * 1103515245 + 12345) & 0x7fffffff;
    return seed / 0x7fffffff;
  }
  function pick(arr) { return arr[Math.floor(rand() * arr.length)]; }

  function buildOrders(n) {
    var orders = [];
    var base = new Date(2024, 0, 1).getTime();
    for (var i = 0; i < n; i++) {
      var cust = pick(CUSTOMERS);
      var status = pick(STATUSES);
      var itemCount = 1 + Math.floor(rand() * 3);
      var items = [];
      var total = 0;
      for (var j = 0; j < itemCount; j++) {
        var p = pick(PRODUCTS);
        var qty = 1 + Math.floor(rand() * 3);
        items.push({ name: p[0], price: p[1], qty: qty });
        total += p[1] * qty;
      }
      var date = new Date(base + Math.floor(rand() * 150) * 86400000);
      orders.push({
        id: "ORD-" + String(1000 + i),
        customer: cust[0],
        email: cust[1],
        date: date,
        status: status,
        items: items,
        total: Math.round(total * 100) / 100
      });
    }
    return orders;
  }

  var ORDERS = buildOrders(42);

  var state = {
    search: "",
    status: "",
    sortKey: "date",
    sortDir: "desc",
    page: 1,
    perPage: 20
  };

  var els = {
    search: document.getElementById("search"),
    status: document.getElementById("status"),
    perPage: document.getElementById("perPage"),
    tbody: document.getElementById("tbody"),
    empty: document.getElementById("empty"),
    count: document.getElementById("count"),
    range: document.getElementById("range"),
    prev: document.getElementById("prev"),
    next: document.getElementById("next"),
    pages: document.getElementById("pages"),
    scrim: document.getElementById("scrim"),
    drawer: document.getElementById("drawer"),
    drawerBody: document.getElementById("drawerBody"),
    drawerClose: document.getElementById("drawerClose"),
    drawerTitle: document.getElementById("drawerTitle")
  };

  var lastFocused = null;

  function money(n) {
    return "$" + n.toFixed(2);
  }
  function fmtDate(d) {
    var y = d.getFullYear();
    var m = String(d.getMonth() + 1).padStart(2, "0");
    var day = String(d.getDate()).padStart(2, "0");
    return y + "-" + m + "-" + day;
  }

  function filtered() {
    var q = state.search.trim().toLowerCase();
    var out = ORDERS.filter(function (o) {
      if (state.status && o.status !== state.status) return false;
      if (!q) return true;
      return (
        o.id.toLowerCase().indexOf(q) !== -1 ||
        o.customer.toLowerCase().indexOf(q) !== -1 ||
        o.email.toLowerCase().indexOf(q) !== -1
      );
    });
    var key = state.sortKey;
    var dir = state.sortDir === "asc" ? 1 : -1;
    out.sort(function (a, b) {
      var av = a[key], bv = b[key];
      if (key === "date") { av = av.getTime(); bv = bv.getTime(); }
      if (typeof av === "string") {
        return av.localeCompare(bv) * dir;
      }
      return (av - bv) * dir;
    });
    return out;
  }

  function render() {
    var rows = filtered();
    var total = rows.length;
    var perPage = state.perPage;
    var pageCount = Math.max(1, Math.ceil(total / perPage));
    if (state.page > pageCount) state.page = pageCount;
    if (state.page < 1) state.page = 1;

    var start = (state.page - 1) * perPage;
    var pageRows = rows.slice(start, start + perPage);

    els.tbody.innerHTML = "";
    pageRows.forEach(function (o) {
      var tr = document.createElement("tr");
      tr.innerHTML =
        '<td>' + o.id + '</td>' +
        '<td>' + escapeHtml(o.customer) + '</td>' +
        '<td>' + fmtDate(o.date) + '</td>' +
        '<td><span class="badge" data-status="' + o.status + '">' + capitalize(o.status) + '</span></td>' +
        '<td class="num">' + money(o.total) + '</td>' +
        '<td><button class="row-btn" type="button" data-id="' + o.id + '">View</button></td>';
      els.tbody.appendChild(tr);
    });

    els.empty.hidden = total !== 0;

    els.count.textContent = total + (total === 1 ? " order" : " orders");
    var from = total === 0 ? 0 : start + 1;
    var to = Math.min(start + perPage, total);
    els.range.textContent = "Showing " + from + "–" + to + " of " + total;

    els.prev.disabled = state.page <= 1;
    els.next.disabled = state.page >= pageCount;

    els.pages.innerHTML = "";
    for (var p = 1; p <= pageCount; p++) {
      var b = document.createElement("button");
      b.type = "button";
      b.textContent = String(p);
      if (p === state.page) b.setAttribute("aria-current", "page");
      b.addEventListener("click", (function (pg) {
        return function () { state.page = pg; render(); };
      })(p));
      els.pages.appendChild(b);
    }

    // Reflect sort state on headers.
    document.querySelectorAll(".sort-btn").forEach(function (btn) {
      var key = btn.getAttribute("data-key");
      if (key === state.sortKey) {
        btn.setAttribute("aria-sort", state.sortDir === "asc" ? "ascending" : "descending");
        btn.querySelector(".arrow").textContent = state.sortDir === "asc" ? "▲" : "▼";
      } else {
        btn.removeAttribute("aria-sort");
        btn.querySelector(".arrow").textContent = "▲";
      }
    });
  }

  function capitalize(s) { return s.charAt(0).toUpperCase() + s.slice(1); }
  function escapeHtml(s) {
    return String(s).replace(/[&<>"']/g, function (c) {
      return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c];
    });
  }

  function openDrawer(id) {
    var o = ORDERS.filter(function (x) { return x.id === id; })[0];
    if (!o) return;
    lastFocused = document.activeElement;

    var itemsHtml = o.items.map(function (it) {
      return '<tr><td>' + escapeHtml(it.name) + '</td><td class="num">' + it.qty +
        '</td><td class="num">' + money(it.price) + '</td><td class="num">' + money(it.price * it.qty) + '</td></tr>';
    }).join("");

    els.drawerTitle.textContent = o.id;
    els.drawerBody.innerHTML =
      '<dl>' +
        '<dt>Customer</dt><dd>' + escapeHtml(o.customer) + '</dd>' +
        '<dt>Email</dt><dd>' + escapeHtml(o.email) + '</dd>' +
        '<dt>Date</dt><dd>' + fmtDate(o.date) + '</dd>' +
        '<dt>Status</dt><dd><span class="badge" data-status="' + o.status + '">' + capitalize(o.status) + '</span></dd>' +
      '</dl>' +
      '<div class="items">' +
        '<h3>Items</h3>' +
        '<table><thead><tr><th scope="col">Product</th><th scope="col" class="num">Qty</th>' +
        '<th scope="col" class="num">Price</th><th scope="col" class="num">Subtotal</th></tr></thead>' +
        '<tbody>' + itemsHtml + '</tbody></table>' +
        '<div class="total"><span>Total</span><span>' + money(o.total) + '</span></div>' +
      '</div>';

    els.drawer.classList.add("open");
    els.scrim.classList.add("open");
    els.drawer.setAttribute("aria-hidden", "false");
    els.drawerClose.focus();
  }

  function closeDrawer() {
    els.drawer.classList.remove("open");
    els.scrim.classList.remove("open");
    els.drawer.setAttribute("aria-hidden", "true");
    if (lastFocused && typeof lastFocused.focus === "function") lastFocused.focus();
  }

  // Events
  els.search.addEventListener("input", function () {
    state.search = els.search.value;
    state.page = 1;
    render();
  });
  els.status.addEventListener("change", function () {
    state.status = els.status.value;
    state.page = 1;
    render();
  });
  els.perPage.addEventListener("change", function () {
    state.perPage = parseInt(els.perPage.value, 10);
    state.page = 1;
    render();
  });
  els.prev.addEventListener("click", function () {
    if (state.page > 1) { state.page--; render(); }
  });
  els.next.addEventListener("click", function () {
    state.page++; render();
  });

  document.querySelectorAll(".sort-btn").forEach(function (btn) {
    btn.addEventListener("click", function () {
      var key = btn.getAttribute("data-key");
      if (state.sortKey === key) {
        state.sortDir = state.sortDir === "asc" ? "desc" : "asc";
      } else {
        state.sortKey = key;
        state.sortDir = "asc";
      }
      render();
    });
  });

  els.tbody.addEventListener("click", function (e) {
    var btn = e.target.closest(".row-btn");
    if (btn) openDrawer(btn.getAttribute("data-id"));
  });

  els.drawerClose.addEventListener("click", closeDrawer);
  els.scrim.addEventListener("click", closeDrawer);
  document.addEventListener("keydown", function (e) {
    if (e.key === "Escape" && els.drawer.classList.contains("open")) closeDrawer();
  });

  render();
})();
</script>
</body>
