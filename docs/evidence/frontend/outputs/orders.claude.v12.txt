<script>
(() => {
  "use strict";

  const STATUS = {
    failed:     { label: "Payment failed", rank: 0 },
    pending:    { label: "Pending",        rank: 1 },
    processing: { label: "Processing",     rank: 2 },
    shipped:    { label: "Shipped",        rank: 3 },
    delivered:  { label: "Delivered",      rank: 4 },
    refunded:   { label: "Refunded",       rank: 5 },
    cancelled:  { label: "Cancelled",      rank: 6 }
  };
  const NEXT = {
    processing: { to: "shipped",   label: "Mark as shipped" },
    shipped:    { to: "delivered", label: "Mark as delivered" }
  };
  const SORTERS = {
    number:   (a, b) => a.number - b.number,
    customer: (a, b) => a.name.localeCompare(b.name),
    date:     (a, b) => a.date - b.date,
    items:    (a, b) => a.count - b.count,
    total:    (a, b) => a.total - b.total,
    status:   (a, b) => STATUS[a.status].rank - STATUS[b.status].rank
  };
  const FIRST_DIR = { customer: "asc", status: "asc" };

  const fmtDate = new Intl.DateTimeFormat("en-US", { month: "short", day: "numeric", year: "numeric", timeZone: "UTC" });
  const fmtDateTime = new Intl.DateTimeFormat("en-US", { month: "short", day: "numeric", year: "numeric", hour: "numeric", minute: "2-digit", timeZone: "UTC" });
  const fmtMoney = new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" });
  const money = cents => fmtMoney.format(cents / 100);
  const esc = s => String(s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const $ = s => document.querySelector(s);

  const els = {
    table: $("#orders"), rows: $("#rows"), q: $("#q"), status: $("#status"), clear: $("#clear"),
    range: $("#range"), pager: $("#pager"), size: $("#size"),
    scrim: $("#scrim"), drawer: $("#drawer"), title: $("#drawer-title"), body: $("#drawer-body"),
    close: $("#drawer-close"), dismiss: $("#drawer-dismiss"), next: $("#drawer-next"), toast: $("#toast")
  };

  const state = { q: "", status: "", sort: { key: "date", dir: "desc" }, page: 1, size: 10, loading: true, error: false };
  let orders = [];
  let openId = null;
  let lastFocus = null;

  /* Sample data */
  function generateOrders() {
    const rng = seed => () => (seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0) / 4294967296;
    const r = rng(20261003);
    const pick = a => a[Math.floor(r() * a.length)];
    const FIRST = ["Amara", "Jonas", "Priya", "Mateo", "Hana", "Liam", "Sofia", "Kenji", "Noor", "Tobias", "Elif", "Rafael", "Ingrid", "Samir", "Chloe", "Dmitri"];
    const LAST = ["Okafor", "Lindqvist", "Raman", "Alvarez", "Sato", "Brennan", "Rossi", "Tanaka", "Haddad", "Weber", "Yilmaz", "Costa", "Berg", "Khan", "Dubois", "Petrov"];
    const CATALOG = [
      ["Canvas tote bag", 2800], ["Ceramic pour-over set", 5400], ["Wool throw blanket", 9600],
      ["Steel water bottle", 2400], ["Desk lamp, matte black", 6800], ["Linen apron", 3200],
      ["Notebook, 3-pack", 1800], ["Cast iron skillet", 4500], ["Beeswax candle", 1600], ["Walnut cutting board", 5800]
    ];
    const STREETS = ["Alder St", "Birch Ave", "Cedar Ln", "Dawn Rd", "Elm Ct", "Fir Way"];
    const CITIES = ["Portland, OR", "Austin, TX", "Madison, WI", "Asheville, NC", "Boise, ID", "Duluth, MN"];

    let t = Date.now() - 20 * 60e3;
    return Array.from({ length: 48 }, (_, i) => {
      t -= Math.round((5 + r() * 10) * 36e5);
      const status = i < 5 ? (r() < 0.4 ? "failed" : "pending")
        : i < 10 ? "processing"
        : i < 16 ? "shipped"
        : r() < 0.08 ? "refunded" : r() < 0.14 ? "cancelled" : "delivered";

      const n = 1 + Math.floor(r() * 3);
      const used = new Set();
      const lines = [];
      while (lines.length < n) {
        const k = Math.floor(r() * CATALOG.length);
        if (used.has(k)) continue;
        used.add(k);
        lines.push({ name: CATALOG[k][0], unit: CATALOG[k][1], qty: 1 + Math.floor(r() * 3) });
      }
      const subtotal = lines.reduce((s, l) => s + l.unit * l.qty, 0);
      const shipping = subtotal >= 7500 ? 0 : 600;
      const tax = Math.round(subtotal * 0.08);
      const first = pick(FIRST), last = pick(LAST);
      const number = 10547 - i;
      const city = pick(CITIES);
      const o = {
        id: String(number), number, label: "SD-" + number, date: t, status, lines, subtotal, shipping, tax,
        total: subtotal + shipping + tax,
        count: lines.reduce((s, l) => s + l.qty, 0),
        name: first + " " + last,
        email: (first + "." + last).toLowerCase() + "@example.com",
        address: (100 + Math.floor(r() * 900)) + " " + pick(STREETS) + ", " + city,
        stamps: {}
      };
      o.hay = [o.label, o.name, o.email, city, ...lines.map(l => l.name)].join(" ").toLowerCase();
      return o;
    });
  }

  function load() {
    state.loading = true;
    state.error = false;
    render();
    return new Promise((res, rej) => setTimeout(() => {
      try { res(generateOrders()); } catch (e) { rej(e); }
    }, 450))
      .then(d => { orders = d; })
      .catch(() => { state.error = true; })
      .then(() => { state.loading = false; render(); updateStats(); });
  }

  /* Table */
  function visible() {
    const q = state.q.trim().toLowerCase();
    const { key, dir } = state.sort;
    const m = dir === "asc" ? 1 : -1;
    return orders
      .filter(o => (!state.status || o.status === state.status) && (!q || o.hay.includes(q)))
      .sort((a, b) => m * SORTERS[key](a, b) || b.number - a.number);
  }

  const rowHtml = o => `<tr data-id="${o.id}"${o.id === openId ? ' class="is-active"' : ""}>
    <td><button type="button" class="idlink" data-open="${o.id}" aria-haspopup="dialog" aria-label="View order ${esc(o.label)}">${esc(o.label)}</button><span class="sub-cust">${esc(o.name)}</span></td>
    <td class="col-cust"><span class="cust-name">${esc(o.name)}</span><span class="cust-mail">${esc(o.email)}</span></td>
    <td class="col-date">${fmtDate.format(o.date)}</td>
    <td class="num col-items">${o.count}</td>
    <td class="num">${money(o.total)}</td>
    <td>${badge(o.status)}</td>
  </tr>`;

  const badge = s => `<span class="badge" data-status="${s}"><span class="dot" aria-hidden="true"></span>${STATUS[s].label}</span>`;

  const skeletonHtml = () => Array.from({ length: 8 }, () => `<tr aria-hidden="true">
    <td><span class="sk s"></span></td><td class="col-cust"><span class="sk"></span></td>
    <td class="col-date"><span class="sk"></span></td><td class="col-items"><span class="sk s"></span></td>
    <td><span class="sk s"></span></td><td><span class="sk"></span></td></tr>`).join("");

  function pageList(cur, total) {
    const nums = [...new Set([1, total, cur - 1, cur, cur + 1])].filter(n => n >= 1 && n <= total).sort((a, b) => a - b);
    const out = [];
    nums.forEach((n, i) => { if (i && n - nums[i - 1] > 1) out.push("…"); out.push(n); });
    return out;
  }

  function renderPager(total, pages) {
    if (!total) { els.pager.innerHTML = ""; return; }
    const cur = state.page;
    let h = `<button class="btn" type="button" data-nav="prev" data-page="${cur - 1}"${cur <= 1 ? " disabled" : ""}>Previous</button>`;
    h += `<span class="pg-of">Page ${cur} of ${pages}</span>`;
    pageList(cur, pages).forEach(n => {
      h += n === "…"
        ? `<span class="pg-gap" aria-hidden="true">…</span>`
        : `<button class="btn pg-num" type="button" data-nav="${n}" data-page="${n}" aria-label="Page ${n}"${n === cur ? ' aria-current="page"' : ""}>${n}</button>`;
    });
    h += `<button class="btn" type="button" data-nav="next" data-page="${cur + 1}"${cur >= pages ? " disabled" : ""}>Next</button>`;
    els.pager.innerHTML = h;
  }

  function syncSortHeaders() {
    document.querySelectorAll("th [data-sort]").forEach(b => {
      const active = b.dataset.sort === state.sort.key;
      b.parentElement.setAttribute("aria-sort", active ? (state.sort.dir === "asc" ? "ascending" : "descending") : "none");
    });
  }

  function render() {
    syncSortHeaders();
    els.table.setAttribute("aria-busy", state.loading ? "true" : "false");
    els.clear.hidden = !(state.q || state.status);

    if (state.loading) {
      els.rows.innerHTML = skeletonHtml();
      els.range.textContent = "Loading orders…";
      els.pager.innerHTML = "";
      return;
    }
    if (state.error) {
      els.rows.innerHTML = `<tr><td colspan="6" class="state-cell"><h3>Orders could not be loaded</h3><p>The request failed. Check the connection and try again.</p><button class="btn btn-primary" type="button" data-retry>Retry</button></td></tr>`;
      els.range.textContent = "";
      els.pager.innerHTML = "";
      return;
    }

    const list = visible();
    const pages = Math.max(1, Math.ceil(list.length / state.size));
    state.page = Math.min(state.page, pages);
    const start = (state.page - 1) * state.size;
    const slice = list.slice(start, start + state.size);

    if (!list.length) {
      els.rows.innerHTML = `<tr><td colspan="6" class="state-cell"><h3>No orders match</h3><p>Nothing matches the current search and status filter.</p><button class="btn" type="button" data-clear>Clear filters</button></td></tr>`;
      els.range.textContent = "0 orders";
    } else {
      els.rows.innerHTML = slice.map(rowHtml).join("");
      els.range.textContent = `Showing ${start + 1}–${start + slice.length} of ${list.length} order${list.length === 1 ? "" : "s"}`;
    }
    renderPager(list.length, pages);
  }

  function updateStats() {
    const c = s => orders.filter(o => o.status === s).length;
    $("#st-action").textContent = c("pending") + c("failed");
    $("#st-processing").textContent = c("processing");
    $("#st-shipped").textContent = c("shipped");
    $("#st-delivered").textContent = c("delivered");
    $("#st-refunded").textContent = c("refunded");
  }

  /* Drawer */
  const byId = id => orders.find(o => o.id === id);

  function events(o) {
    const m = 60e3, h = 60 * m, d = 24 * h, s = o.status;
    const st = (k, fallback) => o.stamps[k] || fallback;
    const ev = [{ label: "Order placed", at: o.date }];
    if (s === "pending") ev.push({ label: "Awaiting payment" });
    else if (s === "failed") ev.push({ label: "Payment failed", at: o.date + 5 * m });
    else {
      ev.push({ label: "Payment confirmed", at: o.date + 4 * m });
      if (s === "cancelled") ev.push({ label: "Cancelled", at: o.date + 3 * h });
      else if (s === "refunded") ev.push({ label: "Refunded", at: o.date + 2 * d });
      else if (s !== "processing") {
        ev.push({ label: "Shipped", at: st("shipped", o.date + d) });
        if (s === "delivered") ev.push({ label: "Delivered", at: st("delivered", o.date + 3 * d) });
      }
    }
    return ev;
  }

  function fillDrawer() {
    const o = byId(openId);
    if (!o) return;
    els.title.textContent = o.label;
    els.body.innerHTML = `
      <div class="d-top">${badge(o.status)}<span class="fine">Placed ${fmtDateTime.format(o.date)} UTC</span></div>
      <section>
        <h3>Customer</h3>
        <p class="d-strong">${esc(o.name)}</p>
        <p><a href="mailto:${esc(o.email)}">${esc(o.email)}</a></p>
        <p class="muted">Ship to ${esc(o.address)}</p>
      </section>
      <section>
        <h3>Items</h3>
        <table class="d-items">
          <thead><tr><th scope="col">Item</th><th scope="col" class="num">Qty</th><th scope="col" class="num">Total</th></tr></thead>
          <tbody>${o.lines.map(l => `<tr><td>${esc(l.name)}</td><td class="num">${l.qty}</td><td class="num">${money(l.unit * l.qty)}</td></tr>`).join("")}</tbody>
        </table>
      </section>
      <section>
        <h3>Payment</h3>
        <dl class="d-sum">
          <div><dt>Subtotal</dt><dd>${money(o.subtotal)}</dd></div>
          <div><dt>Shipping</dt><dd>${o.shipping ? money(o.shipping) : "Free"}</dd></div>
          <div><dt>Tax</dt><dd>${money(o.tax)}</dd></div>
          <div class="total"><dt>Total</dt><dd>${money(o.total)}</dd></div>
        </dl>
      </section>
      <section>
        <h3>Timeline</h3>
        <ol class="timeline">${events(o).map(e => `<li><span class="t-label">${e.label}</span>${e.at ? `<span class="t-time">${fmtDateTime.format(e.at)} UTC</span>` : ""}</li>`).join("")}</ol>
      </section>`;
    const nx = NEXT[o.status];
    els.next.disabled = !nx;
    els.next.textContent = nx ? nx.label : "No next step";
  }

  function markActive() {
    els.rows.querySelectorAll("tr[data-id]").forEach(tr => tr.classList.toggle("is-active", tr.dataset.id === openId));
  }

  function openDrawer(id, trigger) {
    openId = id;
    lastFocus = trigger || document.activeElement;
    fillDrawer();
    els.drawer.classList.add("is-open");
    els.scrim.classList.add("is-open");
    document.body.classList.add("lock");
    markActive();
    els.close.focus();
  }

  function closeDrawer() {
    if (!openId) return;
    const id = openId;
    openId = null;
    els.drawer.classList.remove("is-open");
    els.scrim.classList.remove("is-open");
    document.body.classList.remove("lock");
    markActive();
    const back = els.rows.querySelector(`[data-open="${id}"]`) || (lastFocus && lastFocus.isConnected ? lastFocus : els.q);
    back.focus();
  }

  let toastTimer;
  function toast(msg) {
    els.toast.textContent = msg;
    els.toast.hidden = false;
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => { els.toast.hidden = true; }, 3500);
  }

  els.next.addEventListener("click", () => {
    const o = byId(openId);
    const nx = o && NEXT[o.status];
    if (!nx) return;
    els.next.disabled = true;
    els.next.textContent = "Saving…";
    els.next.setAttribute("aria-busy", "true");
    setTimeout(() => {
      o.status = nx.to;
      o.stamps[nx.to] = Date.now();
      els.next.removeAttribute("aria-busy");
      if (openId === o.id) {
        fillDrawer();
        (els.next.disabled ? els.close : els.next).focus();
      }
      render();
      updateStats();
      toast(`${o.label} marked as ${STATUS[nx.to].label.toLowerCase()}.`);
    }, 500);
  });

  els.drawer.addEventListener("keydown", e => {
    if (e.key !== "Tab") return;
    const f = [...els.drawer.querySelectorAll("button:not([disabled]), a[href]")];
    if (!f.length) return;
    const first = f[0], last = f[f.length - 1];
    if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
    else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
  });
  document.addEventListener("keydown", e => { if (e.key === "Escape" && openId) closeDrawer(); });
  els.scrim.addEventListener("click", closeDrawer);
  els.close.addEventListener("click", closeDrawer);
  els.dismiss.addEventListener("click", closeDrawer);

  /* Controls */
  let debounce;
  els.q.addEventListener("input", () => {
    clearTimeout(debounce);
    debounce = setTimeout(() => { state.q = els.q.value; state.page = 1; render(); }, 120);
  });
  els.status.addEventListener("change", () => { state.status = els.status.value; state.page = 1; render(); });
  els.size.addEventListener("change", () => { state.size = +els.size.value; state.page = 1; render(); });

  function clearFilters() {
    state.q = ""; state.status = ""; state.page = 1;
    els.q.value = ""; els.status.value = "";
    render();
    els.q.focus();
  }
  els.clear.addEventListener("click", clearFilters);

  document.querySelector("thead").addEventListener("click", e => {
    const b = e.target.closest("[data-sort]");
    if (!b || state.loading || state.error) return;
    const key = b.dataset.sort;
    if (state.sort.key === key) state.sort.dir = state.sort.dir === "asc" ? "desc" : "asc";
    else state.sort = { key, dir: FIRST_DIR[key] || "desc" };
    state.page = 1;
    render();
  });

  els.rows.addEventListener("click", e => {
    if (e.target.closest("[data-retry]")) { load(); return; }
    if (e.target.closest("[data-clear]")) { clearFilters(); return; }
    const tr = e.target.closest("tr[data-id]");
    if (tr) openDrawer(tr.dataset.id, tr.querySelector("[data-open]"));
  });

  els.pager.addEventListener("click", e => {
    const b = e.target.closest("button[data-page]");
    if (!b || b.disabled) return;
    const nav = b.dataset.nav;
    state.page = +b.dataset.page;
    render();
    (els.pager.querySelector(`[data-nav="${nav}"]:not([disabled])`) || els.pager.querySelector('[aria-current="page"]')).focus();
  });

  load();
})();
</script>
