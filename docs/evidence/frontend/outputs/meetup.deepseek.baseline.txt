    (function() {
      const form = document.getElementById('rsvpForm');
      const feedback = document.getElementById('formFeedback');

      // simple validation + fake submission
      form.addEventListener('submit', function(e) {
        e.preventDefault();

        const name = document.getElementById('name').value.trim();
        const email = document.getElementById('email').value.trim();

        // reset feedback
        feedback.textContent = '';
        feedback.style.color = '#7ee0b0';

        if (!name) {
          feedback.textContent = 'Please enter your full name.';
          feedback.style.color = '#f28b82';
          return;
        }

        if (!email) {
          feedback.textContent = 'Please enter your email address.';
          feedback.style.color = '#f28b82';
          return;
        }

        // very basic email pattern
        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
          feedback.textContent = 'Please enter a valid email address.';
          feedback.style.color = '#f28b82';
          return;
        }

        // simulate successful RSVP (no backend)
        feedback.textContent = `Thanks, ${name}! Your RSVP is confirmed. See you at Jakarta Rust Night. 🦀`;
        feedback.style.color = '#7ee0b0';

        // optional: reset form fields (but keep name/email for demo? we reset all)
        // form.reset();  // uncomment if you want to clear after success
        // but we keep values so user sees what they submitted; we just show message.
        // However, to avoid double-submit confusion, we could disable button briefly.
        const btn = form.querySelector('button[type="submit"]');
        btn.disabled = true;
        btn.textContent = 'RSVP sent ✓';
        btn.style.background = '#3f5a4b';
        btn.style.color = '#b0e0c0';
        btn.style.cursor = 'default';

        // re-enable after 3 seconds (for demo purposes)
        setTimeout(() => {
          btn.disabled = false;
          btn.textContent = 'Confirm RSVP →';
          btn.style.background = '#e06c4b';
          btn.style.color = '#0b0e14';
          btn.style.cursor = 'pointer';
        }, 3000);
      });

      // small extra: if user clicks checkbox, no action needed.
      // but we can add a tiny touch: show lightning talk hint in message? not necessary.
    })();
