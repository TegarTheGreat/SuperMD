    var ok = f.el.type === "email" ? /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v) : v !== "";
    f.el.setAttribute("aria-invalid", ok ? "false" : "true");
    f.err.textContent = ok ? "" : f.msg;
    return ok;
  }

  fields.forEach(function (f) {
    f.el.addEventListener("blur", function () { if (f.el.value !== "") check(f); });
    f.el.addEventListener("input", function () {
      if (f.el.getAttribute("aria-invalid") === "true") check(f);
    });
  });

  function setBusy(busy) {
    button.disabled = busy;
    button.textContent = busy ? "Sending..." : "Reserve my seat";
    form.setAttribute("aria-busy", busy ? "true" : "false");
    Array.prototype.forEach.call(form.elements, function (el) {
      if (el !== button) el.disabled = busy;
    });
  }

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    formError.textContent = "";

    var firstBad = null;
    fields.forEach(function (f) { if (!check(f) && !firstBad) firstBad = f.el; });
    if (firstBad) { firstBad.focus(); return; }

    var payload = {
      name: fields[0].el.value.trim(),
      email: fields[1].el.value.trim(),
      level: fields[2].el.value,
      diet: document.getElementById("diet").value,
      lightning: document.getElementById("lightning").checked
    };

    function done() {
      document.getElementById("confirm-name").textContent = payload.name.split(" ")[0];
      document.getElementById("confirm-email").textContent = payload.email;
      form.hidden = true;
      confirmBox.hidden = false;
      confirmBox.focus();
    }

    if (!ENDPOINT) { done(); return; }

    setBusy(true);
    fetch(ENDPOINT, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    }).then(function (res) {
      if (!res.ok) throw new Error("HTTP " + res.status);
      done();
    }).catch(function () {
      setBusy(false);
      formError.textContent = "Your RSVP did not go through. Check your connection and try again.";
    });
  });
})();
</script>
</body>
</html>
