  (function () {
    var form = document.getElementById('rsvp-form');
    var status = document.getElementById('form-status');
    var btn = document.getElementById('submit-btn');

    function setStatus(message, kind) {
      status.textContent = message;
      status.className = 'form-status ' + kind;
    }

    form.addEventListener('submit', function (event) {
      event.preventDefault();

      var name = form.name.value.trim();
      var email = form.email.value.trim();
      var attending = form.attending.value;

      if (!name) {
        setStatus('Enter your name.', 'err');
        form.name.focus();
        return;
      }
      if (!email || email.indexOf('@') === -1) {
        setStatus('Enter a valid email address.', 'err');
        form.email.focus();
        return;
      }
      if (!attending) {
        setStatus('Choose whether you are attending.', 'err');
        form.attending.focus();
        return;
      }

      btn.disabled = true;
      btn.textContent = 'Sending…';

      // Placeholder: replace with a fetch() to your RSVP endpoint.
      window.setTimeout(function () {
        btn.disabled = false;
        btn.textContent = 'Send RSVP';
        if (attending === 'no') {
          setStatus('Thanks for letting us know. See you next month.', 'ok');
        } else {
          setStatus('Thanks, ' + name + '. Your RSVP is recorded. We\'ll email the details to ' + email + '.', 'ok');
        }
        form.reset();
      }, 400);
    });
  })();
</script>
