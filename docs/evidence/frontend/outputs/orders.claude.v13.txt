  const out = [];
  let prev = 0;
  for (const p of keep) { if (p - prev > 1) out.push('…'); out.push(p); prev = p; }
  return out;
}

function renderPager(total, pages, start, count) {
  $('#range').textContent = total ? `Showing ${start + 1}–${start + count} of ${total}` : 'No orders';
  const btn = (label, page, disabled, current, aria) =>
    h('button', { type: 'button', class: 'pg', 'data-page': page, disabled, 'aria-current': current ? 'page' : false, 'aria-label': aria }, label);
  const nav = $('#pages');
  nav.replaceChildren(
    btn('Previous', state.page - 1, state.page === 1),
    ...pageList(state.page, pages).map(p => p === '…'
      ? h('span', { class: 'gap', 'aria-hidden': 'true' }, '…')
      : btn(String(p), p, false, p === state.page, `Page ${p}`)),
    btn('Next', state.page + 1, state.page === pages));
}

function render() {
  const list = visible();
  const pages = Math.max(1, Math.ceil(list.length / state.size));
  state.page = Math.min(state.page, pages);
  const start = (state.page - 1) * state.size;
  const slice = list.slice(start, start + state.size);

  $('#st-action').textContent = list.filter(o => o.status === 'Pending' || o.status === 'Paid').length;
  $('#st-shown').textContent = list.length;
  $('#st-value').textContent = money(list.filter(o => o.status !== 'Cancelled' && o.status !== 'Refunded').reduce((s, o) => s + o.total, 0));

  tbody.replaceChildren(...(slice.length ? slice.map(rowFor) : [
    h('tr', {}, h('td', { colspan: 6, class: 'empty' },
      h('p', {}, 'No orders match these filters.'),
      h('button', { type: 'button', class: 'btn', 'data-clear': true }, 'Clear filters')))
  ]));

  renderPager(list.length, pages, start, slice.length);
  document.querySelectorAll('thead th').forEach(th => {
    if (th.dataset.col === state.sortKey) th.setAttribute('aria-sort', state.dir === 'asc' ? 'ascending' : 'descending');
    else th.removeAttribute('aria-sort');
  });
  $('#clear').disabled = !(state.q || state.status);
  live.textContent = `${list.length} orders, page ${state.page} of ${pages}`;
}

function clearFilters() {
  state.q = ''; state.status = ''; state.page = 1;
  $('#q').value = ''; $('#status').value = '';
  render();
  $('#q').focus();
}

$('#q').addEventListener('input', e => { state.q = e.target.value; state.page = 1; render(); });
$('#status').addEventListener('change', e => { state.status = e.target.value; state.page = 1; render(); });
$('#size').addEventListener('change', e => { state.size = +e.target.value; state.page = 1; render(); });
document.addEventListener('click', e => { if (e.target.closest('[data-clear]')) clearFilters(); });

$('thead').addEventListener('click', e => {
  const b = e.target.closest('.sort');
  if (!b) return;
  const k = b.dataset.key;
  if (state.sortKey === k) state.dir = state.dir === 'asc' ? 'desc' : 'asc';
  else { state.sortKey = k; state.dir = ['date', 'total', 'items'].includes(k) ? 'desc' : 'asc'; }
  state.page = 1;
  render();
});

$('#pages').addEventListener('click', e => {
  const b = e.target.closest('button[data-page]');
  if (!b || b.disabled) return;
  state.page = +b.dataset.page;
  render();
  $('#pages [aria-current="page"]').focus();
});
