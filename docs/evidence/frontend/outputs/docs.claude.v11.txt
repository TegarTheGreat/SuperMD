    <h1 id="overview" style="margin-top:0">Create an invoice</h1>
    <div class="endpoint">
      <span class="method">POST</span>
      <span class="path">/v1/invoices</span>
    </div>
    <p>Creates an invoice for an existing customer. By default the invoice is finalized immediately: its number is assigned, its totals are locked, and its status is <code>open</code>. Set <code>auto_finalize</code> to <code>false</code> to create a <code>draft</code> you can still edit.</p>
    <p>Base URL: <code>https://api.example.com</code> <span class="opt">[confirm: production base URL]</span></p>

    <ul class="meta">
      <li><b>Content type</b>application/json</li>
      <li><b>Success</b>201 Created</li>
      <li><b>Idempotent</b>Yes, with <code>Idempotency-Key</code></li>
      <li><b>Rate limit</b><span class="opt">[confirm: requests per minute]</span></li>
    </ul>

    <h2 id="authentication">Authentication</h2>
    <p>Send a secret API key as a bearer token. Keys are server-side credentials; do not embed them in browser or mobile code. A missing or invalid key returns <code>401</code>, and a valid key without permission to create invoices returns <code>403</code>.</p>
    <pre style="background:var(--code-bg);border-radius:8px;margin-bottom:20px"><code style="background:none;border:0;color:var(--code-text);padding:0">Authorization: Bearer YOUR_API_KEY</code></pre>

    <h2 id="idempotency">Idempotency</h2>
    <p>Send an <code>Idempotency-Key</code> on every create call. If a network failure or timeout hides the response, repeating the request with the same key and body returns the original result instead of creating a second invoice. Reusing a key with a different body returns <code>409</code>. Retention period for keys: <span class="opt">[confirm: key lifetime]</span>.</p>
    <div class="note">Retry <code>429</code> and <code>5xx</code> responses with exponential backoff and the same key. Do not retry <code>4xx</code> responses other than <code>429</code> without changing the request.</div>

        else if (e.key === 'Home') next = tabs[0];
        else if (e.key === 'End') next = tabs[tabs.length - 1];

      });
    });
  });

  // Copy: copies the visible panel; falls back to a hidden textarea where the Clipboard API is unavailable.
  function copyText(text) {
    if (navigator.clipboard && window.isSecureContext) return navigator.clipboard.writeText(text);
    return new Promise(function (resolve, reject) {
      var ta = document.createElement('textarea');
      ta.value = text; ta.setAttribute('readonly', ''); ta.style.position = 'fixed'; ta.style.opacity = '0';
      document.body.appendChild(ta); ta.select();
      try { document.execCommand('copy') ? resolve() : reject(); } catch (err) { reject(err); }
      document.body.removeChild(ta);
    });
  }
  document.querySelectorAll('.codeblock').forEach(function (block) {
    var btn = block.querySelector('.copy');
    var live = document.createElement('span');
    live.setAttribute('role', 'status');
    live.style.cssText = 'position:absolute;left:-9999px';
    block.appendChild(live);
    btn.addEventListener('click', function () {
      var pre = Array.prototype.slice.call(block.querySelectorAll('pre')).filter(function (p) {
        return !p.closest('[hidden]');
      })[0];
      btn.disabled = true;
      copyText(pre.textContent).then(function () {
        btn.textContent = 'Copied'; live.textContent = 'Copied to clipboard';
      }, function () {
        btn.textContent = 'Copy failed'; live.textContent = 'Copy failed';
      }).then(function () {
        setTimeout(function () { btn.textContent = 'Copy'; btn.disabled = false; live.textContent = ''; }, 1600);
      });
    });
  });

  // Scroll spy: marks the nav link for the section nearest the top of the viewport.
  var links = Array.prototype.slice.call(document.querySelectorAll('nav.sidebar a[href^="#"]'));
  var map = {};
  links.forEach(function (a) { map[a.getAttribute('href').slice(1)] = a; });
  var targets = Object.keys(map).map(function (id) { return document.getElementById(id); }).filter(Boolean);
  function setCurrent(id) {
    links.forEach(function (a) { a.removeAttribute('aria-current'); });
    if (map[id]) map[id].setAttribute('aria-current', 'true');
  }
  if ('IntersectionObserver' in window) {
    var visible = {};
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) { visible[e.target.id] = e.isIntersecting; });
      for (var i = 0; i < targets.length; i++) {
        if (visible[targets[i].id]) { setCurrent(targets[i].id); return; }
      }
    }, { rootMargin: '0px 0px -70% 0px' });
    targets.forEach(function (t) { io.observe(t); });
  }
  links.forEach(function (a) {
    a.addEventListener('click', function () { setCurrent(a.getAttribute('href').slice(1)); });
  });
  setCurrent('overview');
})();
