    <section id="overview">
      <h1>Create an invoice</h1>
      <div class="endpoint">
        <span class="method">POST</span>
        <span class="url"><span>https://api.example.com</span>/v1/invoices</span>
      </div>
      <p>Creates an invoice for an existing customer. The invoice starts in <code>open</code> status and the customer is notified according to <code>collection_method</code>. Set <code>draft</code> to <code>true</code> to create it in <code>draft</code> status instead; nothing is sent until a draft is finalized.</p>
      <p>Amounts are integers in the currency's smallest unit, so <code>120000</code> in <code>usd</code> is $1,200.00.</p>
      <dl class="facts">
        <div><dt>Auth</dt><dd>Bearer API key</dd></div>
        <div><dt>Scope</dt><dd>invoices:write</dd></div>
        <div><dt>Success</dt><dd>201 Created</dd></div>
        <div><dt>Retries</dt><dd>Idempotency-Key header</dd></div>
      </dl>
      <p class="note">Sample content: hosts, IDs, amounts, field limits and error codes are illustrative. Replace before launch.</p>
    </section>

    <section id="authentication">
      <h2>Authentication and headers</h2>
      <p>Send the API key as a bearer token. Keys are secret; call this endpoint from a server, never from browser code.</p>
      <table class="table">
        <thead><tr><th scope="col">Header</th><th scope="col">Required</th><th scope="col">Description</th></tr></thead>
        <tbody>
          <tr>
            <td class="name full">Authorization</td>
            <td><span class="req">Required</span></td>
            <td class="full"><code>Bearer &lt;API key&gt;</code>. The key needs the <code>invoices:write</code> scope.</td>
          </tr>
          <tr>
            <td class="name full">Content-Type</td>
            <td><span class="req">Required</span></td>
            <td class="full"><code>application/json</code></td>
          </tr>
          <tr>
            <td class="name full">Idempotency-Key</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">A unique string per logical request, such as a UUID. Retrying with the same key and body returns the original response. The same key with a different body returns <code>409</code>. Send it on every call.</td>
          </tr>
        </tbody>
      </table>
    </section>

    <section id="parameters">
      <h2>Body parameters</h2>
      <table class="table">
        <thead><tr><th scope="col">Parameter</th><th scope="col">Type</th><th scope="col">Required</th><th scope="col">Description</th></tr></thead>
        <tbody>
          <tr>
            <td class="name full">customer_id</td>
            <td class="type">string</td>
            <td><span class="req">Required</span></td>
            <td class="full">ID of an existing customer, such as <code>cus_8f3k2d</code>. An unknown ID returns <code>customer_not_found</code>.</td>
          </tr>
          <tr>
            <td class="name full">currency</td>
            <td class="type">string</td>
            <td><span class="req">Required</span></td>
            <td class="full">Lowercase three-letter ISO 4217 code, such as <code>usd</code>. Applies to every line item.</td>
          </tr>
          <tr>
            <td class="name full">line_items</td>
            <td class="type">array of objects</td>
            <td><span class="req">Required</span></td>
            <td class="full">One or more items to bill. Each item takes the fields below.</td>
          </tr>
          <tr>
            <td class="name full">line_items[].description</td>
            <td class="type">string</td>
            <td><span class="req">Required</span></td>
            <td class="full">Text shown on the invoice row.</td>
          </tr>
          <tr>
            <td class="name full">line_items[].unit_amount</td>
            <td class="type">integer</td>
            <td><span class="req">Required</span></td>
            <td class="full">Price per unit in the currency's smallest unit (cents for <code>usd</code>). Zero-decimal currencies such as <code>jpy</code> use whole units.</td>
          </tr>
          <tr>
            <td class="name full">line_items[].quantity</td>
            <td class="type">integer</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">Units billed. Default <code>1</code>.</td>
          </tr>
          <tr>
            <td class="name full">due_date</td>
            <td class="type">string</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">Date as <code>YYYY-MM-DD</code>, today or later. Default: 30 days after creation.</td>
          </tr>
          <tr>
            <td class="name full">collection_method</td>
            <td class="type">string</td>
            <td><span class="opt">Optional</span></td>
            <td class="full"><code>send_invoice</code> emails the customer a payment link. <code>charge_automatically</code> charges the customer's default payment method. Default <code>send_invoice</code>.</td>
          </tr>
          <tr>
            <td class="name full">draft</td>
            <td class="type">boolean</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">When <code>true</code>, the invoice is created as <code>draft</code> and nothing is sent. Default <code>false</code>.</td>
          </tr>
          <tr>
            <td class="name full">memo</td>
            <td class="type">string</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">Free text printed at the bottom of the invoice.</td>
          </tr>
          <tr>
            <td class="name full">metadata</td>
            <td class="type">object</td>
            <td><span class="opt">Optional</span></td>
            <td class="full">String key-value pairs stored with the invoice and returned unchanged. Not shown to the customer.</td>
          </tr>
        </tbody>
      </table>

    <section id="errors">
      <h2>Errors</h2>
      <p>Failures return a standard HTTP status and a JSON body with an <code>error</code> object. <code>param</code> names the offending field when one applies. Include <code>request_id</code> when you contact support.</p>
      <table class="table">
        <thead><tr><th scope="col">Status</th><th scope="col">Code</th><th scope="col">Meaning</th><th scope="col">What to do</th></tr></thead>
        <tbody>
          <tr>
            <td class="name">400</td>
            <td class="name"><code>invalid_request</code></td>
            <td class="full">The body is not valid JSON, or a required parameter is missing.</td>
            <td class="full">Fix the payload. Do not retry it unchanged.</td>
          </tr>
          <tr>
            <td class="name">401</td>
            <td class="name"><code>unauthenticated</code></td>
            <td class="full">The API key is missing, malformed, or revoked.</td>
            <td class="full">Send a valid key in <code>Authorization</code>.</td>
          </tr>
          <tr>
            <td class="name">403</td>
            <td class="name"><code>forbidden</code></td>
            <td class="full">The key lacks the <code>invoices:write</code> scope.</td>
            <td class="full">Use a key that has the scope.</td>
          </tr>
          <tr>
            <td class="name">404</td>
            <td class="name"><code>customer_not_found</code></td>
            <td class="full">No customer matches <code>customer_id</code>.</td>
            <td class="full">Check the ID.</td>
          </tr>
          <tr>
            <td class="name">409</td>
            <td class="name"><code>idempotency_conflict</code></td>
            <td class="full">The <code>Idempotency-Key</code> was already used with a different body.</td>
            <td class="full">Generate a new key for the new request.</td>
          </tr>
          <tr>
            <td class="name">422</td>

(() => {
  const esc = (s) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

  // JSON highlighting; without JS the blocks stay plain text.
  const token = /("(?:\\.|[^"\\])*")(\s*:)?|\b(?:true|false|null)\b|-?\b\d+(?:\.\d+)?\b/g;
  document.querySelectorAll('code[data-lang="json"]').forEach((el) => {
    const src = el.textContent;
    let out = "", last = 0, m;
    token.lastIndex = 0;
    while ((m = token.exec(src))) {
      out += esc(src.slice(last, m.index));
      out += m[1]
        ? `<span class="${m[2] ? "tok-k" : "tok-s"}">${esc(m[1])}</span>${esc(m[2] || "")}`
        : `<span class="tok-n">${esc(m[0])}</span>`;
      last = token.lastIndex;
    }
    el.innerHTML = out + esc(src.slice(last));
  });

  // Copy buttons need the async clipboard API (secure contexts only).
  if (navigator.clipboard && navigator.clipboard.writeText) {
    document.querySelectorAll("figure.code").forEach((fig) => {
      const btn = document.createElement("button");
      btn.type = "button";
      btn.className = "copy";
      btn.textContent = "Copy";
      btn.setAttribute("aria-live", "polite");
      btn.addEventListener("click", async () => {
        try {
          await navigator.clipboard.writeText(fig.querySelector("code").textContent);
          btn.textContent = "Copied";
        } catch {
          btn.textContent = "Copy failed";
        }
        setTimeout(() => { btn.textContent = "Copy"; }, 1600);
      });
      fig.querySelector("figcaption").append(btn);
    });
  }

  // Tabs: without JS every panel stays visible with its own label.
  document.querySelectorAll("[data-tabs]").forEach((root, i) => {
    const panels = [...root.querySelectorAll(":scope > .panel")];
    const list = document.createElement("div");
    list.className = "tablist";
    list.setAttribute("role", "tablist");
    list.setAttribute("aria-label", root.dataset.tabs);

    const tabs = panels.map((panel, n) => {
      const tab = document.createElement("button");
      tab.type = "button";
      tab.className = "tab";
      tab.id = `tab-${i}-${n}`;
      tab.textContent = panel.dataset.tab;
      tab.setAttribute("role", "tab");
      tab.setAttribute("aria-controls", `panel-${i}-${n}`);
      panel.id = `panel-${i}-${n}`;
      panel.setAttribute("role", "tabpanel");
      panel.setAttribute("aria-labelledby", tab.id);
      tab.addEventListener("click", () => select(n));
      list.append(tab);
      return tab;
    });

    function select(n, focus) {
      tabs.forEach((t, k) => {
        t.setAttribute("aria-selected", k === n);
        t.tabIndex = k === n ? 0 : -1;
        panels[k].hidden = k !== n;
      });
      if (focus) tabs[n].focus();
    }

    list.addEventListener("keydown", (e) => {
      const cur = tabs.indexOf(document.activeElement);
      if (cur < 0) return;
      const target = { ArrowRight: cur + 1, ArrowLeft: cur - 1, Home: 0, End: tabs.length - 1 }[e.key];
      if (target === undefined) return;
      e.preventDefault();
      select((target + tabs.length) % tabs.length, true);
    });

    root.prepend(list);
    root.classList.add("is-enhanced");
    select(0);
  });

  // Highlight the current section in the sidebar.
  const links = [...document.querySelectorAll(".toc a")];
  const byId = new Map(links.map((a) => [a.hash.slice(1), a]));
  if ("IntersectionObserver" in window) {
    const io = new IntersectionObserver((entries) => {
      entries.forEach((e) => {
        if (!e.isIntersecting) return;
        links.forEach((a) => a.removeAttribute("aria-current"));
        const link = byId.get(e.target.id);
        if (link) link.setAttribute("aria-current", "true");
      });
    }, { rootMargin: "-72px 0px -65% 0px" });
    document.querySelectorAll("main section[id]").forEach((s) => io.observe(s));
  }
})();
