(function () {
  var form = document.getElementById('rsvp-form');
  var status = document.getElementById('status');
  var KEY = 'jrn-rsvp-2026-10';

  function setError(id, msg) {
    var input = document.getElementById(id);
    document.getElementById(id + '-err').textContent = msg;
    if (msg) input.setAttribute('aria-invalid', 'true');
    else input.removeAttribute('aria-invalid');
  }

  function validate() {
    var ok = true;
    var name = form.name.value.trim();
    var email = form.email.value.trim();

    setError('name', name.length < 2 ? 'Please enter your name.' : '');
    if (name.length < 2) ok = false;

    var emailOk = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    setError('email', emailOk ? '' : 'Please enter a valid email address.');
    if (!emailOk) ok = false;

    setError('coc', form.coc.checked ? '' : 'Please agree to the code of conduct.');
    if (!form.coc.checked) ok = false;

    return ok;
  }

  ['name', 'email'].forEach(function (id) {
    document.getElementById(id).addEventListener('input', function () {
      if (this.getAttribute('aria-invalid')) validate();
    });
  });
  form.coc.addEventListener('change', function () { if (this.checked) setError('coc', ''); });

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    status.classList.remove('show');
    if (!validate()) {
      var first = form.querySelector('[aria-invalid="true"]');
      if (first) first.focus();
      return;
    }

    var data = {
      name: form.name.value.trim(),
      email: form.email.value.trim(),
      attend: form.attend.value,
      level: form.level.value,
      diet: form.diet.value,
      notes: form.notes.value.trim(),
      savedAt: new Date().toISOString()
    };

    // This page has no backend. To collect RSVPs, POST `data` to your own
    // endpoint here. For now the RSVP is only kept in this browser.
    try { localStorage.setItem(KEY, JSON.stringify(data)); } catch (err) {}

    status.textContent = '';
    var msg = document.createElement('span');
    msg.textContent = 'Thanks, ' + data.name.split(' ')[0] + '! Your RSVP is recorded (';
    var code = document.createElement('code');
    code.textContent = data.attend === 'yes' ? 'status: going' : 'status: maybe';
    msg.appendChild(code);
    msg.appendChild(document.createTextNode('). See you on 22 October.'));
    status.appendChild(msg);
    status.classList.add('show');
    status.focus();
    form.reset();
  });
})();
</script>
