      // ---------- STATE ----------
      let currentPage = 1;
      const pageSize = 8;
      let sortKey = 'id';
      let sortDir = 'asc'; // 'asc' or 'desc'
      let searchTerm = '';
      let statusFilter = 'all';

      const tbody = document.getElementById('ordersBody');
      const searchInput = document.getElementById('searchInput');

      // ---------- HELPERS ----------
      function formatCurrency(value) {
        return '$' + value.toFixed(2);
      }

      function getStatusClass(status) {
        return status.toLowerCase();
      }

      // filter + sort + paginate
      function getFilteredSortedOrders() {
        let filtered = orders.filter(order => {
          const matchesSearch = searchTerm === '' ||
            order.customer.toLowerCase().includes(searchTerm.toLowerCase()) ||
            order.id.toLowerCase().includes(searchTerm.toLowerCase());
          const matchesStatus = statusFilter === 'all' || order.status === statusFilter;
          return matchesSearch && matchesStatus;
        });

        // sort
        filtered.sort((a, b) => {
          let valA = a[sortKey];
          let valB = b[sortKey];
          if (sortKey === 'total') {
            valA = Number(valA);
            valB = Number(valB);
          } else if (sortKey === 'date') {
            valA = new Date(valA);
            valB = new Date(valB);
          } else {
            valA = String(valA).toLowerCase();
            valB = String(valB).toLowerCase();
          }
          if (valA < valB) return sortDir === 'asc' ? -1 : 1;
          if (valA > valB) return sortDir === 'asc' ? 1 : -1;
          return 0;
        });

        return filtered;
      }

      // render table + pagination
      function render() {
        const filtered = getFilteredSortedOrders();
        const total = filtered.length;
        const totalPages = Math.max(1, Math.ceil(total / pageSize));
        if (currentPage > totalPages) currentPage = totalPages;
        if (currentPage < 1) currentPage = 1;

        const start = (currentPage - 1) * pageSize;
        const end = Math.min(start + pageSize, total);
        const pageOrders = filtered.slice(start, end);

        // update order count badge
        orderCountSpan.textContent = total;

        // render rows
        if (pageOrders.length === 0) {
          tbody.innerHTML = `<tr><td colspan="6" class="empty-message">No orders match your criteria</td></tr>`;
        } else {
          tbody.innerHTML = pageOrders.map(order => {

            return `<tr>

              <td>${order.customer}</td>
              <td>${order.date}</td>
              <td>${formatCurrency(order.total)}</td>
              <td><span class="status-badge ${statusClass}">${order.status}</span></td>
              <td><button class="btn-detail" data-id="${order.id}">Details</button></td>
            </tr>`;
          }).join('');
        }

        // update page info
        if (total === 0) {
          pageInfo.textContent = 'Showing 0–0 of 0';
        } else {
          pageInfo.textContent = `Showing ${start + 1}–${end} of ${total}`;

        // render pagination buttons

        // update sort indicators

          if (indicator) {
            if (key === sortKey) {
              th.classList.add('active-sort');
              indicator.textContent = sortDir === 'asc' ? '↑' : '↓';
            } else {
              indicator.textContent = '↕';
            }
          }
        });
      }

      function renderPagination(totalPages) {
        let html = '';
        // prev
        html += `<button class="page-btn" data-page="${currentPage - 1}" ${currentPage === 1 ? 'disabled' : ''}>←</button>`;

        // page numbers (show up to 5 pages around current)
        const maxVisible = 5;
        let startPage = Math.max(1, currentPage - Math.floor(maxVisible / 2));
        let endPage = Math.min(totalPages, startPage + maxVisible - 1);
        if (endPage - startPage + 1 < maxVisible) {
          startPage = Math.max(1, endPage - maxVisible + 1);
        }

        for (let i = startPage; i <= endPage; i++) {
          html += `<button class="page-btn ${i === currentPage ? 'active' : ''}" data-page="${i}">${i}</button>`;
        }

        // next
        html += `<button class="page-btn" data-page="${currentPage + 1}" ${currentPage === totalPages ? 'disabled' : ''}>→</button>`;

        pageControls.innerHTML = html;

        // attach listeners to page buttons

            if (!isNaN(page) && page >= 1 && page <= totalPages && page !== currentPage) {
              currentPage = page;
              render();
            }
          });
        });

      });

      // sortable columns
      document.querySelectorAll('th[data-key]').forEach(th => {
        th.addEventListener('click', () => {
          const key = th.dataset.key;
          if (sortKey === key) {
            sortDir = sortDir === 'asc' ? 'desc' : 'asc';
          } else {
            sortKey = key;
            sortDir = 'asc';
          }
          currentPage = 1;
          render();
        });
      });

      // detail buttons (event delegation)
      tbody.addEventListener('click', (e) => {
        const btn = e.target.closest('.btn-detail');
        if (btn) {
          const orderId = btn.dataset.id;
          openDrawer(orderId);
        }
      });

      // close drawer
      drawerCloseBtn.addEventListener('click', closeDrawer);
      drawerOverlay.addEventListener('click', closeDrawer);
      document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && drawer.classList.contains('open')) {
          closeDrawer();
        }
      });

      // ---------- INITIAL RENDER ----------
      render();
    })();
  </script>
</body>
</html>
