  const form = document.getElementById('form');
  const status = document.getElementById('status');
  const EMAIL = document.querySelector('[data-email]').textContent.trim();

  function setError(id, msg) {
    const input = document.getElementById(id);
    const err = document.getElementById('e-' + id.slice(2));
    err.textContent = msg;
    if (msg) input.setAttribute('aria-invalid', 'true'); else input.removeAttribute('aria-invalid');
    return !msg;
  }

  form.addEventListener('submit', e => {
    e.preventDefault();
    const d = new FormData(form);
    const email = String(d.get('email')).trim();
    const results = [
      setError('f-name', String(d.get('name')).trim() ? '' : 'Enter your name.'),
      setError('f-email', /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ? '' : 'Enter a valid email address.'),
      setError('f-msg', String(d.get('message')).trim().length >= 10 ? '' : 'Add a few details (at least 10 characters).')
    ];
    if (results.includes(false)) {
      status.textContent = 'Fix the highlighted fields.';
      form.querySelector('[aria-invalid="true"]').focus();
      return;
    }
    const body = [
      'Name: ' + d.get('name'),
      'Reply-to: ' + email,
      'Type: ' + d.get('type'),
      'Date: ' + (d.get('date') || 'not set'),
      '',
      d.get('message')
    ].join('\n');
    const href = 'mailto:' + EMAIL + '?subject=' + encodeURIComponent('Booking enquiry: ' + d.get('type')) + '&body=' + encodeURIComponent(body);
    status.textContent = 'Opening your email app. If nothing opens, write to ' + EMAIL + ' directly.';
    window.location.href = href;
  });

  form.addEventListener('input', e => {
    if (e.target.getAttribute('aria-invalid')) setError(e.target.id, '');
  });
</script>
</body>
</html>
