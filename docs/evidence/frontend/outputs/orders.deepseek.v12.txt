  function formatCurrency(n) {
    return "$" + n.toFixed(2);
  }

  function formatDate(iso) {
    var parts = iso.split("-");
    var d = new Date(Date.UTC(+parts[0], +parts[1] - 1, +parts[2]));
    return d.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric", timeZone: "UTC" });
  }

  function statusLabel(s) {
    return s.charAt(0).toUpperCase() + s.slice(1);
  }

  function getFiltered() {
    var q = state.search.trim().toLowerCase();
    var rows = ORDERS.filter(function (o) {
      if (state.status && o.status !== state.status) return false;
      if (!q) return true;
      return o.id.toLowerCase().indexOf(q) !== -1 ||
             o.customer.toLowerCase().indexOf(q) !== -1 ||
             o.email.toLowerCase().indexOf(q) !== -1;
    });

    var key = state.sortKey;
    var dir = state.sortDir === "asc" ? 1 : -1;
    rows.sort(function (a, b) {
      var av = a[key], bv = b[key];
      if (key === "total") return (av - bv) * dir;
      if (key === "date") return (av < bv ? -1 : av > bv ? 1 : 0) * dir;
      av = String(av).toLowerCase();
      bv = String(bv).toLowerCase();
      return (av < bv ? -1 : av > bv ? 1 : 0) * dir;
    });
    return rows;
  }

  function render() {
    var rows = getFiltered();
    var totalPages = Math.max(1, Math.ceil(rows.length / PAGE_SIZE));
    if (state.page > totalPages) state.page = totalPages;
    var start = (state.page - 1) * PAGE_SIZE;
    var pageRows = rows.slice(start, start + PAGE_SIZE);

    els.body.innerHTML = "";

    if (pageRows.length === 0) {
      var tr = document.createElement("tr");
      var td = document.createElement("td");
      td.colSpan = 5;
      td.className = "empty-state";
      td.innerHTML = "<strong>No orders match</strong>Adjust the search or status filter.";
      tr.appendChild(td);
      els.body.appendChild(tr);
    } else {
      pageRows.forEach(function (o) {
        var tr = document.createElement("tr");
        tr.tabIndex = 0;
        tr.setAttribute("data-id", o.id);
        tr.innerHTML =
          '<td class="mono">' + o.id + '</td>' +
          '<td>' + escapeHtml(o.customer) + '</td>' +
          '<td class="mono">' + formatDate(o.date) + '</td>' +
          '<td><span class="badge ' + o.status + '">' + statusLabel(o.status) + '</span></td>' +
          '<td class="num mono">' + formatCurrency(o.total) + '</td>';
        tr.addEventListener("click", function () { openDrawer(o.id); });
        tr.addEventListener("keydown", function (e) {
          if (e.key === "Enter" || e.key === " ") {
            e.preventDefault();
            openDrawer(o.id);
          }
        });
        els.body.appendChild(tr);
      });
    }

    els.resultCount.textContent = rows.length + (rows.length === 1 ? " order" : " orders");

    renderPagination(totalPages);
    updateSortIndicators();
  }
