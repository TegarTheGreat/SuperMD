    <section id="overview">
      <h1>Create an invoice</h1>
      <p class="lead">Creates a new invoice for an existing customer. Invoices are created in <code>draft</code> status unless <code>finalize</code> is set to <code>true</code>.</p>

      <div class="endpoint">
        <span class="method">POST</span>
        <span class="url"><span>https://api.example.com</span>/v1/invoices</span>
      </div>

      <p>Totals are calculated by the server from the supplied line items, tax rates, and discount. When an invoice is finalized it is assigned a sequential number, locked for editing, and (if <code>send_email</code> is enabled) delivered to the customer.</p>

      <div class="callout">
        <strong>Amounts are in minor units</strong>
        All monetary values are integers in the smallest currency unit (for example, cents for <code>usd</code>). <code>1999</code> means $19.99.
      </div>
    </section>

    <section id="authentication">
      <h2>Authentication</h2>
      <p>Authenticate with a secret API key sent as a Bearer token. Keep keys server-side; never embed them in client code or public repositories.</p>
      <div class="code-block">
        <div class="code-head"><span>Header</span><button class="copy" type="button">Copy</button></div>
        <pre>Authorization: Bearer sk_live_your_api_key</pre>
      </div>
    </section>

            <tr>
              <td class="name"><code>customer_id</code> <span class="badge req">Required</span></td>
              <td class="type">string</td>

            </tr>
            <tr>
              <td class="name"><code>currency</code> <span class="badge req">Required</span></td>
              <td class="type">string</td>

            <tr>
              <td class="name"><code>line_items</code> <span class="badge req">Required</span></td>

              <td class="name"><code>due_date</code> <span class="badge opt">Optional</span></td>
              <td class="type">string (date)</td>
              <td>Payment due date in <code>YYYY-MM-DD</code> format. Must not be in the past. Defaults to 30 days from creation.</td>
            </tr>
            <tr>
              <td class="name"><code>discount_percent</code> <span class="badge opt">Optional</span></td>
              <td class="type">number</td>
              <td>Invoice-level discount from <code>0</code> to <code>100</code>. Applied to the subtotal before tax.</td>
            </tr>
            <tr>
              <td class="name"><code>memo</code> <span class="badge opt">Optional</span></td>
              <td class="type">string</td>
              <td>Note shown to the customer on the invoice. Max 1,000 characters.</td>
            </tr>
            <tr>
              <td class="name"><code>finalize</code> <span class="badge opt">Optional</span></td>
              <td class="type">boolean</td>
              <td>If <code>true</code>, finalize immediately instead of leaving as a draft. Default <code>false</code>.</td>
            </tr>
            <tr>
              <td class="name"><code>send_email</code> <span class="badge opt">Optional</span></td>
              <td class="type">boolean</td>
              <td>Email the invoice to the customer on finalization. Ignored unless <code>finalize</code> is <code>true</code>. Default <code>false</code>.</td>
            </tr>
            <tr>
              <td class="name"><code>metadata</code> <span class="badge opt">Optional</span></td>
              <td class="type">object</td>
              <td>Up to 20 string key–value pairs for your own use. Keys max 40 chars, values max 500 chars.</td>
            </tr>
          </tbody>

(function () {
  // Copy buttons
  document.querySelectorAll('.copy').forEach(function (btn) {
    btn.addEventListener('click', function () {
      var block = btn.closest('.code-block');
      var pane = block.querySelector('pre:not([hidden])');
      var text = pane.textContent;
      var done = function () {
        btn.textContent = 'Copied';
        setTimeout(function () { btn.textContent = 'Copy'; }, 1500);
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(text).then(done, fallback);
      } else {
        fallback();
      }
      function fallback() {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try { document.execCommand('copy'); done(); } catch (e) {}
        document.body.removeChild(ta);
      }
    });
  });

  // Language tabs
  document.querySelectorAll('[data-tabs]').forEach(function (block) {
    var tabs = block.querySelectorAll('.tab');
    var panes = block.querySelectorAll('pre[data-pane]');
    tabs.forEach(function (tab) {
      tab.addEventListener('click', function () {
        tabs.forEach(function (t) { t.setAttribute('aria-selected', String(t === tab)); });
        panes.forEach(function (p) { p.hidden = p.dataset.pane !== tab.dataset.tab; });
      });
    });
  });

  // Scroll spy
  var links = Array.prototype.slice.call(document.querySelectorAll('#nav-links a'));
  var map = {};
  links.forEach(function (a) { map[a.getAttribute('href').slice(1)] = a; });
  var targets = Object.keys(map).map(function (id) { return document.getElementById(id); }).filter(Boolean);

  function setActive(id) {
    links.forEach(function (a) { a.classList.remove('active'); });
    if (map[id]) map[id].classList.add('active');
  }

  function onScroll() {
    var current = targets[0] && targets[0].id;
    var offset = 120;
    for (var i = 0; i < targets.length; i++) {
      if (targets[i].getBoundingClientRect().top - offset <= 0) current = targets[i].id;
    }
    if (window.innerHeight + window.scrollY >= document.body.scrollHeight - 4) {
      current = targets[targets.length - 1].id;
    }
    setActive(current);
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  // Mobile menu
  var toggle = document.getElementById('menu-toggle');
  var navLinks = document.getElementById('nav-links');
  toggle.addEventListener('click', function () {
    var open = navLinks.classList.toggle('open');
    toggle.setAttribute('aria-expanded', String(open));
  });
  links.forEach(function (a) {
    a.addEventListener('click', function () {
      navLinks.classList.remove('open');
      toggle.setAttribute('aria-expanded', 'false');
    });
  });
})();
