      ? "Showing " + startIdx + "–" + endIdx + " of " + totalRows + " orders"
      : "0 orders";
    $("prev").disabled = state.page <= 1;
    $("next").disabled = state.page >= pages;

    var box = $("pages");
    box.textContent = "";
    pageList(state.page, pages).forEach(function (p) {
      if (p === "…") { box.appendChild(el("span", { class: "gap", text: "…", "aria-hidden": "true" })); return; }
      var b = el("button", { class: "btn page-btn", type: "button", text: String(p), "aria-label": "Page " + p });
      if (p === state.page) b.setAttribute("aria-current", "page");
      b.addEventListener("click", function () { state.page = p; render(); });
      box.appendChild(b);
    });
  }

  function render() {
    var rows = getFiltered();
    var pages = Math.max(1, Math.ceil(rows.length / state.size));
    if (state.page > pages) state.page = pages;
    var slice = rows.slice((state.page - 1) * state.size, state.page * state.size);
    renderHead();
    renderBody(slice);
    renderPager(rows.length);
  }

  /* ---------- Drawer ---------- */
  function timeline(o) {
    var t = new Date(o.date).getTime();
    var H = 3600000;
    var steps = [["Order placed", t]];
    switch (o.status) {
      case "Processing": steps.push(["Payment confirmed", t + 0.2 * H]); break;
      case "Shipped":
        steps.push(["Payment confirmed", t + 0.2 * H], ["Shipped", t + 26 * H]); break;
      case "Delivered":
        steps.push(["Payment confirmed", t + 0.2 * H], ["Shipped", t + 26 * H], ["Delivered", t + 98 * H]); break;
      case "Cancelled": steps.push(["Cancelled", t + 5 * H]); break;
      case "Refunded":
        steps.push(["Payment confirmed", t + 0.2 * H], ["Shipped", t + 26 * H], ["Delivered", t + 98 * H], ["Refunded", t + 220 * H]); break;
      default: steps.push(["Awaiting payment", null]);
    }
    return steps;
  }

  function kv(pairs) {
    var dl = el("dl", { class: "kv" });
    pairs.forEach(function (p) {
      dl.appendChild(el("dt", { text: p[0] }));
      var dd = el("dd");
      if (p[1] instanceof Node) dd.appendChild(p[1]); else dd.textContent = p[1];
      dl.appendChild(dd);
    });
    return dl;
  }

  function section(title, node) {
    return el("section", null, [el("h3", { text: title }), node]);
  }

  function fillDrawer(o) {
    $("drawer-title").textContent = o.id;
    $("drawer-meta").textContent = "Placed " + fmtDateTime(new Date(o.date).getTime());
    var body = $("drawer-body");
    body.textContent = "";

    body.appendChild(section("Status", badge(o.status)));

    body.appendChild(section("Customer", kv([
      ["Name", o.customer],
      ["Email", o.email],
      ["Ship to", el("span", null, [
        document.createTextNode(o.address[0]), el("br"), document.createTextNode(o.address[1])
      ])],
      ["Payment", o.payment]
    ])));

    var table = el("table", { class: "items" });
    var thead = el("thead", null, [el("tr", null, [
      el("th", { scope: "col", text: "Item" }),
      el("th", { scope: "col", class: "num", text: "Qty" }),
      el("th", { scope: "col", class: "num", text: "Price" })
    ])]);
    var tbody = el("tbody");
    o.lines.forEach(function (l) {
      tbody.appendChild(el("tr", null, [
        el("td", { text: l.name }),
        el("td", { class: "num", text: String(l.qty) }),
        el("td", { class: "num", text: fmtMoney(l.price * l.qty) })
      ]));
    });
    table.appendChild(thead);
    table.appendChild(tbody);
    var totals = el("dl", { class: "totals" });
    [["Subtotal", o.subtotal], ["Shipping", o.shipping], ["Tax", o.tax]].forEach(function (r) {
      totals.appendChild(el("dt", { text: r[0] }));
      totals.appendChild(el("dd", { text: r[1] ? fmtMoney(r[1]) : "Free" }));
    });
    totals.appendChild(el("dt", { class: "grand", text: "Total" }));
    totals.appendChild(el("dd", { class: "grand", text: fmtMoney(o.total) }));
    body.appendChild(section("Items", el("div", null, [table, totals])));

    var ul = el("ul", { class: "timeline" });
    timeline(o).forEach(function (s) {
      ul.appendChild(el("li", null, [
        document.createTextNode(s[0]),
        el("span", { class: "when", text: s[1] ? fmtDateTime(s[1]) : "Pending" })
      ]));
    });
    body.appendChild(section("Timeline", ul));
    body.scrollTop = 0;
  }

  function openDrawer(id, trigger) {
    var o = ORDERS.filter(function (x) { return x.id === id; })[0];
    if (!o) return;
    lastFocus = trigger || document.activeElement;
    state.selected = id;
    fillDrawer(o);
    $("drawer").inert = false;
    $("main").inert = true;
    $("drawer").classList.add("open");
    $("scrim").classList.add("open");
    Array.prototype.forEach.call($("body").children, function (r) {
      r.classList.toggle("selected", r.getAttribute("data-id") === id);
    });
    $("close").focus();
  }

  function closeDrawer() {
    if (!$("drawer").classList.contains("open")) return;
    $("drawer").classList.remove("open");
    $("scrim").classList.remove("open");
    $("drawer").inert = true;
    $("main").inert = false;
    state.selected = null;
    Array.prototype.forEach.call($("body").children, function (r) { r.classList.remove("selected"); });
    if (lastFocus && document.contains(lastFocus)) lastFocus.focus();
    else {
      var first = $("body").querySelector("tr[tabindex]");
      if (first) first.focus();
    }
  }

  /* ---------- Wiring ---------- */
  function init() {
    var sel = $("status");
    ["All"].concat(STATUSES).forEach(function (s) {
      sel.appendChild(el("option", { value: s, text: s === "All" ? "All statuses" : s }));
    });
    $("size").value = String(state.size);

    $("q").addEventListener("input", function (e) { state.q = e.target.value; state.page = 1; render(); });
    sel.addEventListener("change", function (e) { state.status = e.target.value; state.page = 1; render(); });
    $("size").addEventListener("change", function (e) { state.size = parseInt(e.target.value, 10); state.page = 1; render(); });
    $("prev").addEventListener("click", function () { if (state.page > 1) { state.page--; render(); } });
    $("next").addEventListener("click", function () { state.page++; render(); });
    $("reset").addEventListener("click", function () {
      state.q = ""; state.status = "All"; state.sortKey = "date"; state.sortDir = "desc"; state.page = 1;
      $("q").value = ""; sel.value = "All";
      render();
    });

    $("close").addEventListener("click", closeDrawer);
    $("done").addEventListener("click", closeDrawer);
    $("scrim").addEventListener("click", closeDrawer);
    $("copy").addEventListener("click", function () {
      var btn = this, id = state.selected;
      if (!id) return;
      var done = function (msg) {
        btn.textContent = msg;
        setTimeout(function () { btn.textContent = "Copy order ID"; }, 1500);
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(id).then(function () { done("Copied"); }, function () { done("Copy failed"); });
      } else done("Copy unavailable");
    });

    document.addEventListener("keydown", function (e) {
      var d = $("drawer");
      if (!d.classList.contains("open")) return;
      if (e.key === "Escape") { e.preventDefault(); closeDrawer(); return; }
      if (e.key === "Tab") {
        var f = d.querySelectorAll("button:not([disabled])");
        if (!f.length) return;
        var first = f[0], last = f[f.length - 1];
        if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
        else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
      }
    });

    renderStats();
    render();
  }

  init();
})();
