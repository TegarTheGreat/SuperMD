<script>
(function () {
  var ENDPOINT = ''; // POST target for RSVPs; empty runs a demo that sends nothing
  var form = document.getElementById('rsvp');
  var status = document.getElementById('rsvp-status');
  var btn = document.getElementById('rsvp-submit');
  form.noValidate = true;

  function check(id, test, msg) {
    var input = document.getElementById(id);
    var err = document.getElementById(id + '-err');
    var ok = test(input.value.trim());
    input.setAttribute('aria-invalid', ok ? 'false' : 'true');
    err.textContent = ok ? '' : msg;
    return ok ? null : input;
  }

  function say(text, kind) {
    status.textContent = text;
    status.className = 'status ' + kind;
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var bad = [
      check('f-name', function (v) { return v.length > 0; }, 'Enter your name.'),
      check('f-email', function (v) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(v); }, 'Enter an email like name@example.com.')
    ].filter(Boolean);
    if (bad.length) {
      say('Fix the highlighted fields.', 'is-error');
      bad[0].focus();
      return;
    }

    var data = Object.fromEntries(new FormData(form));
    btn.disabled = true;
    btn.textContent = 'Sending…';
    form.setAttribute('aria-busy', 'true');
    say('', '');

    var send = ENDPOINT
      ? fetch(ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
          .then(function (r) { if (!r.ok) throw new Error(r.status); })
      : new Promise(function (resolve) { setTimeout(resolve, 700); });

    send.then(function () {
      var prefix = ENDPOINT ? '' : 'Demo mode, nothing was sent. ';
      say(prefix + 'RSVP recorded for ' + data.name + '. See you on 22 Oct.', 'is-ok');
      form.reset();
    }).catch(function () {
      say('Could not send your RSVP. Try again in a minute.', 'is-error');
    }).finally(function () {
      btn.disabled = false;
      btn.textContent = 'Send RSVP';
      form.removeAttribute('aria-busy');
    });
  });
})();
</script>
