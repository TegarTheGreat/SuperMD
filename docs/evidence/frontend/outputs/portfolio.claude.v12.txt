(() => {
  const TO = 'hello@example.com';
  const form = document.getElementById('enquiry');
  const status = document.getElementById('form-status');
  const submit = form.querySelector('button[type="submit"]');

  // Native bubbles are replaced by inline messages below.
  form.noValidate = true;

  const rules = {
    name: (v) => (v.trim() ? '' : 'Enter your name.'),
    email: (v) => (/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v.trim()) ? '' : 'Enter an email address like name@example.com.'),
    type: (v) => (v ? '' : 'Choose the kind of shoot.'),
    details: (v) => (v.trim().length >= 10 ? '' : 'Add a few details: place, time and what you need.')
  };

  const setError = (name, message) => {
    document.getElementById(name + '-error').textContent = message;
    form.elements[name].setAttribute('aria-invalid', message ? 'true' : 'false');
    return message;
  };

  document.querySelectorAll('[data-service]').forEach((link) => {
    link.addEventListener('click', () => { form.elements['type'].value = link.dataset.service; });
  });

  const revalidate = (event) => {
    const field = event.target;
    if (rules[field.name] && field.getAttribute('aria-invalid') === 'true') {
      setError(field.name, rules[field.name](field.value));
    }
  };
  form.addEventListener('input', revalidate);
  form.addEventListener('change', revalidate);

  form.addEventListener('submit', (event) => {
    event.preventDefault();
    const invalid = Object.keys(rules).filter((name) => setError(name, rules[name](form.elements[name].value)));
    if (invalid.length) {
      status.textContent = invalid.length === 1 ? 'Fix 1 field and send again.' : 'Fix ' + invalid.length + ' fields and send again.';
      form.elements[invalid[0]].focus();
      return;
    }

    const data = new FormData(form);
    const body = [
      'Name: ' + data.get('name'),
      'Email: ' + data.get('email'),
      'Shoot: ' + data.get('type'),
      'Date: ' + (data.get('date') || 'not decided'),
      '',
      data.get('details')
    ].join('\r\n');

    submit.disabled = true;
    submit.textContent = 'Opening your mail app\u2026';
    status.textContent = 'If nothing opens, write to ' + TO + ' directly.';
    window.location.href = 'mailto:' + TO + '?subject=' + encodeURIComponent('Enquiry: ' + data.get('type')) + '&body=' + encodeURIComponent(body);

    setTimeout(() => {
      submit.disabled = false;
      submit.textContent = 'Send enquiry';
    }, 3000);
  });
})();
</script>
