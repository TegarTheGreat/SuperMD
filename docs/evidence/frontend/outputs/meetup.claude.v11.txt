// Set to the URL that accepts a JSON POST ({name, email, needs, firstTime}).
// Left empty on purpose: without an endpoint the form reports that nothing was sent
// instead of showing a false confirmation.
const RSVP_ENDPOINT = "";

const form = document.getElementById("rsvp-form");
const statusEl = document.getElementById("status");
const submitBtn = document.getElementById("submit");
const submitLabel = document.getElementById("submit-label");
const fields = [
  { el: form.elements.name, err: document.getElementById("name-err"), missing: "Enter your name." },
  { el: form.elements.email, err: document.getElementById("email-err"), missing: "Enter your email address.", invalid: "Enter a valid email address, like name@example.com." },
];

function setStatus(kind, text) {
  statusEl.className = "status " + kind;
  statusEl.textContent = text;
}

function validateField(f) {
  const v = f.el.value.trim();
  let msg = "";
  if (!v) msg = f.missing;
  else if (!f.el.checkValidity()) msg = f.invalid || f.missing;
  f.err.textContent = msg;
  f.el.setAttribute("aria-invalid", msg ? "true" : "false");
  return !msg;
}

fields.forEach(f => {
  f.el.addEventListener("blur", () => validateField(f));
  f.el.addEventListener("input", () => { if (f.el.getAttribute("aria-invalid") === "true") validateField(f); });
});

function setBusy(busy) {
  submitBtn.disabled = busy;
  submitBtn.setAttribute("aria-busy", String(busy));
  submitLabel.textContent = busy ? "Sending…" : "Reserve my spot";
  [...form.elements].forEach(el => { if (el !== submitBtn) el.disabled = busy; });
}

form.addEventListener("submit", async e => {
  e.preventDefault();
  statusEl.className = "status";
  statusEl.textContent = "";

  const results = fields.map(validateField);
  if (results.includes(false)) {
    fields[results.indexOf(false)].el.focus();
    return;
  }

  if (!RSVP_ENDPOINT) {
    setStatus("info", "RSVP is not open yet: no submission endpoint is configured, so nothing was sent.");
    return;
  }

  const payload = {
    name: form.elements.name.value.trim(),
    email: form.elements.email.value.trim(),
    needs: form.elements.needs.value.trim(),
    firstTime: form.elements.firstTime.checked,
  };

  setBusy(true);
  try {
    const res = await fetch(RSVP_ENDPOINT, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    if (!res.ok) throw new Error("HTTP " + res.status);
    form.reset();
    fields.forEach(f => f.el.setAttribute("aria-invalid", "false"));
    setStatus("ok", "RSVP received. See you there.");
  } catch (err) {
    setStatus("fail", "Could not send your RSVP. Check your connection and try again.");
  } finally {
    setBusy(false);
  }
});
</script>
