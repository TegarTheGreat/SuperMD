```jsx
import { useEffect, useMemo, useState } from "react";

const PAGE_SIZE = 10;

export default function UserTable() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [sortDir, setSortDir] = useState("asc");
  const [page, setPage] = useState(1);

  useEffect(() => {
    const controller = new AbortController();

    async function load() {
      try {
        const res = await fetch("/api/users", { signal: controller.signal });
        if (!res.ok) throw new Error(`Request failed: ${res.status}`);
        setUsers(await res.json());
      } catch (err) {
        if (err.name !== "AbortError") setError(err.message);
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }

    load();
    return () => controller.abort();
  }, []);

  const sorted = useMemo(() => {
    const copy = [...users];
    copy.sort((a, b) =>
      sortDir === "asc"
        ? a.name.localeCompare(b.name)
        : b.name.localeCompare(a.name)
    );
    return copy;
  }, [users, sortDir]);

  const totalPages = Math.max(1, Math.ceil(sorted.length / PAGE_SIZE));
  const currentPage = Math.min(page, totalPages);
  const pageUsers = sorted.slice(
    (currentPage - 1) * PAGE_SIZE,
    currentPage * PAGE_SIZE
  );

  if (loading) return <p>Loading users…</p>;
  if (error) return <p role="alert">Error: {error}</p>;
  if (users.length === 0) return <p>No users found.</p>;

  return (
    <div>
      <table>
        <thead>
          <tr>
            <th>
              <button
                type="button"
                onClick={() => setSortDir((d) => (d === "asc" ? "desc" : "asc"))}
              >
                Name {sortDir === "asc" ? "▲" : "▼"}
              </button>
            </th>
            <th>Email</th>
          </tr>
        </thead>
        <tbody>
          {pageUsers.map((user) => (
            <tr key={user.id}>
              <td>{user.name}</td>
              <td>{user.email}</td>
            </tr>
          ))}
        </tbody>
      </table>

      <div>
        <button
          type="button"
          onClick={() => setPage(currentPage - 1)}
          disabled={currentPage === 1}
        >
          Previous
        </button>
        <span>
          {" "}Page {currentPage} of {totalPages}{" "}
        </span>
        <button
          type="button"
          onClick={() => setPage(currentPage + 1)}
          disabled={currentPage === totalPages}
        >
          Next
        </button>
      </div>
    </div>
  );
}
```

Notes:
- Assumes `/api/users` returns a JSON array of objects with `id`, `name`, and `email`. Adjust the columns if your shape differs.
- Sorting and pagination are client-side, so the full list is fetched once. If the list is large, move both to the server with `?sort=name&page=N` query params.
- The fetch is aborted on unmount to avoid state updates on an unmounted component.
- Toggling the sort keeps you on the current page; add `setPage(1)` to the toggle handler if you'd rather reset to the first page.
