```jsx
import { useEffect, useMemo, useState } from 'react';

const PAGE_SIZE = 10;

export default function UserTable() {
  const [users, setUsers] = useState([]);
  const [status, setStatus] = useState('loading'); // 'loading' | 'error' | 'ready'
  const [error, setError] = useState(null);
  const [sortAsc, setSortAsc] = useState(true);
  const [page, setPage] = useState(1);

  useEffect(() => {
    const controller = new AbortController();
    setStatus('loading');
    fetch('/api/users', { signal: controller.signal })
      .then((res) => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json();
      })
      .then((data) => {
        setUsers(data);
        setStatus('ready');
      })
      .catch((err) => {
        if (err.name === 'AbortError') return;
        setError(err);
        setStatus('error');
      });
    return () => controller.abort();
  }, []);

  const sorted = useMemo(() => {
    const copy = [...users];
    copy.sort((a, b) => {
      const cmp = a.name.localeCompare(b.name);
      return sortAsc ? cmp : -cmp;
    });
    return copy;
  }, [users, sortAsc]);

  const pageCount = Math.max(1, Math.ceil(sorted.length / PAGE_SIZE));
  const currentPage = Math.min(page, pageCount);
  const pageRows = sorted.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE);

  if (status === 'loading') return <p>Loading users…</p>;
  if (status === 'error') return <p role="alert">Failed to load users: {error.message}</p>;
  if (users.length === 0) return <p>No users.</p>;

  return (
    <div>
      <table>
        <thead>
          <tr>
            <th scope="col">
              <button
                type="button"
                onClick={() => {
                  setSortAsc((v) => !v);
                  setPage(1);
                }}
                aria-sort={sortAsc ? 'ascending' : 'descending'}
              >
                Name {sortAsc ? '▲' : '▼'}
              </button>
            </th>
            <th scope="col">Email</th>
          </tr>
        </thead>
        <tbody>
          {pageRows.map((user) => (
            <tr key={user.id}>
              <td>{user.name}</td>
              <td>{user.email}</td>
            </tr>
          ))}
        </tbody>
      </table>

      <nav aria-label="Pagination">
        <button type="button" onClick={() => setPage(currentPage - 1)} disabled={currentPage === 1}>
          Previous
        </button>
        <span>
          Page {currentPage} of {pageCount}
        </span>
        <button
          type="button"
          onClick={() => setPage(currentPage + 1)}
          disabled={currentPage === pageCount}
        >
          Next
        </button>
      </nav>
    </div>
  );
}
```

Assumptions I made, since the request didn't specify them:

- **Response shape** is a JSON array of `{ id, name, email }`. If `/api/users` returns `{ data: [...] }` or a paginated envelope, adjust the `.then` that calls `setUsers`.
- **Client-side sort and pagination.** All users are fetched at once, then sorted and sliced in memory. This breaks down past a few thousand rows; for larger sets, push `sort` and `page` to the API as query params and drop the `useMemo`/`slice`.
- **`name` is a non-null string.** `localeCompare` throws on `undefined`; if names can be missing, guard with `(a.name ?? '')`.

Two things worth flagging:

- `aria-sort` belongs on the `<th>`, not the button — I put it there. The button carries the click and the arrow glyph.
- The `AbortController` cleanup prevents a state update after unmount, which matters under React 18 StrictMode's double-invoked effects in development.
