  // Billing toggle
  const sw = document.getElementById('billing');
  const lblM = document.getElementById('lblMonthly');
  const lblY = document.getElementById('lblYearly');
  function setBilling(yearly) {
    sw.setAttribute('aria-checked', yearly);
    lblM.classList.toggle('off', yearly);
    lblY.classList.toggle('off', !yearly);
    const key = yearly ? 'yearly' : 'monthly';
    document.querySelectorAll('.price [data-monthly]').forEach(el => {
      el.textContent = '$' + el.dataset[key];
    });
    document.querySelectorAll('.billed[data-monthly]').forEach(el => {
      el.textContent = el.dataset[key];
    });
  }
  sw.addEventListener('click', () => setBilling(sw.getAttribute('aria-checked') !== 'true'));

  // Interactive demo board: click a card to advance it to the next column
  const cols = document.querySelectorAll('#board .col');
  function updateCounts() {
    cols.forEach(c => { c.querySelector('h4 span').textContent = c.querySelectorAll('.card').length; });
  }
  document.getElementById('board').addEventListener('click', e => {
    const card = e.target.closest('.card');
    if (!card) return;
    const col = card.closest('.col');
    const idx = [...cols].indexOf(col);
    const next = cols[(idx + 1) % cols.length];
    next.appendChild(card);
    card.classList.toggle('done', next === cols[cols.length - 1]);
    updateCounts();
  });

  document.getElementById('year').textContent = new Date().getFullYear();
