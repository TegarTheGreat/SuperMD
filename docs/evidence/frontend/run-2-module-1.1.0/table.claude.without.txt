```jsx
import { useEffect, useMemo, useState } from "react";

const PAGE_SIZE = 10;

export default function UserTable() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [sortDir, setSortDir] = useState("asc");
  const [page, setPage] = useState(1);

  useEffect(() => {
    const controller = new AbortController();

    async function load() {
      try {
        const res = await fetch("/api/users", { signal: controller.signal });
        if (!res.ok) throw new Error(`Request failed: ${res.status}`);
        setUsers(await res.json());
      } catch (err) {
        if (err.name !== "AbortError") setError(err.message);
      } finally {
        if (!controller.signal.aborted) setLoading(false);
      }
    }

    load();
    return () => controller.abort();
  }, []);

  const sorted = useMemo(() => {
    const copy = [...users];
    copy.sort((a, b) =>
      sortDir === "asc"
        ? a.name.localeCompare(b.name)
        : b.name.localeCompare(a.name)
    );
    return copy;
  }, [users, sortDir]);

  const totalPages = Math.max(1, Math.ceil(sorted.length / PAGE_SIZE));
  const currentPage = Math.min(page, totalPages);
  const pageUsers = sorted.slice(
    (currentPage - 1) * PAGE_SIZE,
    currentPage * PAGE_SIZE
  );

  if (loading) return <p>Loading users…</p>;
  if (error) return <p role="alert">Error: {error}</p>;
  if (users.length === 0) return <p>No users found.</p>;

  return (
    <div>
      <table>
        <thead>
          <tr>
            <th
              aria-sort={sortDir === "asc" ? "ascending" : "descending"}
              style={{ cursor: "pointer" }}
              onClick={() => {
                setSortDir((d) => (d === "asc" ? "desc" : "asc"));
                setPage(1);
              }}
            >
              Name {sortDir === "asc" ? "▲" : "▼"}
            </th>
            <th>Email</th>
          </tr>
        </thead>
        <tbody>
          {pageUsers.map((user) => (
            <tr key={user.id}>
              <td>{user.name}</td>
              <td>{user.email}</td>
            </tr>
          ))}
        </tbody>
      </table>

      <div style={{ marginTop: 12, display: "flex", gap: 8, alignItems: "center" }}>
        <button
          onClick={() => setPage(currentPage - 1)}
          disabled={currentPage === 1}
        >
          Previous
        </button>
        <span>
          Page {currentPage} of {totalPages}
        </span>
        <button
          onClick={() => setPage(currentPage + 1)}
          disabled={currentPage === totalPages}
        >
          Next
        </button>
      </div>
    </div>
  );
}
```

Notes:
- It assumes `/api/users` returns a JSON array of objects shaped like `{ id, name, email }`. Adjust the columns if your payload differs.
- Sorting and pagination happen client-side, which suits small to medium lists. For large datasets, move both to the server (e.g. `/api/users?page=2&sort=name:asc`) and refetch on change.
- Clicking the Name header toggles ascending and descending order and resets to page 1.
