    });
  }

    sw.setAttribute('aria-checked', sw.getAttribute('aria-checked') !== 'true');

  document.getElementById('lbl-annual').addEventListener('click', () => { sw.setAttribute('aria-checked', 'true'); renderBilling(); });

  // Interactive demo board: click a card to advance it
  const cols = ['todo', 'doing', 'done'];
  const board = document.getElementById('board');
  function updateCounts() {
    board.querySelectorAll('.col').forEach(col => {
      col.querySelector('h3 span').textContent = col.querySelectorAll('.card').length;
    });
  }
  board.addEventListener('click', e => {
    const card = e.target.closest('.card');
    if (!card) return;
    const col = card.parentElement;
    const next = cols[(cols.indexOf(col.dataset.col) + 1) % cols.length];
    card.classList.toggle('done', next === 'done');
    board.querySelector('[data-col="' + next + '"]').appendChild(card);
    updateCounts();
  });

  // Footer year
  document.getElementById('year').textContent = new Date().getFullYear();
